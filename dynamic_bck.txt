  type: "call" | "message" | "notification" | "music";
  color: string;
  textColor: string;
  icon: () => any;
  line1: string;
  line2: string;
};

interface DynamicIslandProps {
  totalUnreadCount?: number;
  isPublicProfile?: boolean;
}

export default function DynamicIsland(props: DynamicIslandProps) {
  const [currentIndex, setCurrentIndex] = createSignal(0);
  const [isAnimating, setIsAnimating] = createSignal(false);

  let intervalId: number;

  onMount(() => {
    intervalId = window.setInterval(() => {
      setIsAnimating(true);
      setTimeout(() => {
        setCurrentIndex((prev) => (prev + 1) % states.length);
        setIsAnimating(false);
      }, 300);
    }, 2000);
  });

  onCleanup(() => {
    if (intervalId) clearInterval(intervalId);
  });

  const currentState = () => states[currentIndex()];

  return (
    <div class="flex flex-col items-center w-full">
      <div
        class="relative overflow-hidden rounded-2xl shadow-lg transition-all duration-500 ease-out w-12"
        style={{
          "background-color": currentState().color,
          transform: isAnimating() ? "scale(0.92)" : "scale(1)",
          opacity: isAnimating() ? "0.7" : "1"
        }}
      >
        <div class="py-3 px-2 flex flex-col items-center gap-2">
          {/* Animated Icon */}
          <div
            class="transition-all duration-300"
            style={{
              color: currentState().textColor,
              transform: isAnimating() ? "rotate(90deg) scale(0.7)" : "rotate(0deg) scale(1)"
            }}
          >
            {currentState().icon()}
          </div>

          {/* Vertical Text Content */}
          <div
            class="flex flex-col items-center gap-0.5 transition-all duration-300"
            style={{
              transform: isAnimating() ? "translateY(-5px)" : "translateY(0)",
              opacity: isAnimating() ? "0" : "1"
            }}
          >
            {/* Line 1 - Vertical */}
            <div class="flex flex-col items-center">
              {currentState().line1.split('').map((char: string) => (
                <span
                  class="text-[9px] font-bold leading-tight"
                  style={{ color: currentState().textColor }}
                >
                  {char}
                </span>
              ))}
            </div>
            
            {/* Separator Dot */}
            <div
              class="w-1 h-1 rounded-full my-0.5"
              style={{ "background-color": currentState().textColor, opacity: "0.5" }}
            />
            
            {/* Line 2 - Vertical */}
            <div class="flex flex-col items-center">
              {currentState().line2.split('').map((char: string) => (
                <span
                  class="text-[9px] font-medium leading-tight"
                  style={{ color: currentState().textColor }}
                >
                  {char}
                </span>
              ))}
            </div>
          </div>

          {/* Pulse Animation for Call */}
          <Show when={currentState().type === "call"}>
            <div class="mt-1">
              <div
                class="w-1.5 h-1.5 rounded-full animate-pulse"
                style={{ "background-color": currentState().textColor }}
              />
            </div>
          </Show>

          {/* Music Bars Animation */}
          <Show when={currentState().type === "music"}>
            <div class="flex items-end gap-0.5 h-3 mt-1">
              <div
                class="w-0.5 bg-white rounded-full animate-pulse"
                style={{ height: "60%", "animation-delay": "0ms" }}
              />
              <div
                class="w-0.5 bg-white rounded-full animate-pulse"
                style={{ height: "100%", "animation-delay": "150ms" }}
              />
              <div
                class="w-0.5 bg-white rounded-full animate-pulse"
                style={{ height: "40%", "animation-delay": "300ms" }}
              />
            </div>
          </Show>
        </div>

        {/* Shimmer Effect - Vertical */}
        <div
          class="absolute inset-0 opacity-20 pointer-events-none"
          style={{
            background: `linear-gradient(180deg, transparent, ${currentState().textColor}, transparent)`,
            transform: isAnimating() ? "translateY(100%)" : "translateY(-100%)",
            transition: "transform 0.6s ease-in-out"
          }}
        />
      </div>
    </div>
  );
}
